from sqlalchemy.orm import Session

from app.database import get_db
from app.models import User
from app.services.speaker_service import extract_embedding
from app.utils.encryption import encrypt_data
from app.services.audio_preprocessing import preprocess_audio


router = APIRouter(
    prefix="/register",
    tags=["Voice Registration"]
)


@router.post("/")
async def register_user(
    username: str = Form(...),
    email: str = Form(...),
    language: str = Form(...),
    file: UploadFile = File(...),
    db: Session = Depends(get_db)
):

    # Check whether username already exists
    existing_user = db.query(User).filter(
        User.username == username
    ).first()

    if existing_user:
        raise HTTPException(
            status_code=400,
            detail="Username already exists"
        )

    # Check whether email already exists
    existing_email = db.query(User).filter(
        User.email == email
    ).first()

    if existing_email:
        raise HTTPException(
            status_code=400,
            detail="Email already exists"
        )

    # Validate language
    allowed_languages = ["en", "hi", "mr"]

    if language not in allowed_languages:
        raise HTTPException(
            status_code=400,
            detail="Language must be en, hi or mr"
        )

    # Create temp directory
    os.makedirs("temp", exist_ok=True)

    file_path = os.path.join(
        "temp",
        file.filename
    )

    # Save uploaded audio temporarily
    with open(file_path, "wb") as buffer:
        buffer.write(await file.read())

    try:
        # Preprocess audio before extracting speaker embedding
        processed_file_path = preprocess_audio(file_path)

        # Extract ECAPA-TDNN voice embedding
        embedding = extract_embedding(file_path)

        # Convert embedding to bytes
        embedding_bytes =( embedding.detach()
                          .cpu()
                          .numpy()
                          .tobytes())

        # Encrypt embedding
        encrypted_embedding = encrypt_data(
            embedding_bytes
        )

        # Create database user
        user = User(
            username=username,
            email=email,
            language=language,
            encrypted_embedding=encrypted_embedding
        )

        db.add(user)
        db.commit()
        db.refresh(user)

        return {
            "message": "User registered successfully",
            "user_id": user.id,
            "username": user.username,
            "language": user.language
        }

    finally:


        # Delete temporary audio
        if os.path.exists(file_path):
            os.remove(file_path)

        if "processed_file_path" in locals():
            if os.path.exists(processed_file_path):
                os.remove(processed_file_path)

import os

from fastapi import (

from sqlalchemy.orm import Session

from app.utils.encryption import decrypt_data
from app.config import settings
from app.services.audio_preprocessing import preprocess_audio


router = APIRouter(
    prefix="/login",
    tags=["Voice Login"]
)


@router.post("/")
async def voice_login(
    username: str = Form(...),
    challenge: str = Form(...),
    file: UploadFile = File(...),
    db: Session = Depends(get_db)
):

    # -----------------------------------
    # 1. Find registered user
    # -----------------------------------

    user = db.query(User).filter(
        User.username == username
    ).first()

    if not user:
        raise HTTPException(
            status_code=404,
            detail="User not found"
        )

    # -----------------------------------
    # 2. Save audio temporarily
    # -----------------------------------

    os.makedirs("temp", exist_ok=True)

    file_path = os.path.join(
        "temp",
        file.filename
    )

    with open(file_path, "wb") as buffer:
        buffer.write(await file.read())

    try:

        # -----------------------------------
        # 3. LIVENESS CHECK
        # -----------------------------------
        processed_file_path = preprocess_audio(file_path)

        transcription = transcribe_audio(processed_file_path)

        spoken_text = transcription["text"]

        liveness_passed = verify_challenge(
            challenge,
            spoken_text
        )

        if not liveness_passed:

            return {
                "authenticated": False,
                "liveness": False,
                "speaker_match": False,
                "message": "Liveness check failed",
                "spoken_text": spoken_text
            }

        # -----------------------------------
        # 4. ECAPA-TDNN SPEAKER VERIFICATION
        # -----------------------------------

        login_embedding = extract_embedding(
            processed_file_path
          
        )

        # -----------------------------------
        # 5. Decrypt stored embedding
        # -----------------------------------

        decrypted_data = decrypt_data(
            user.encrypted_embedding
        )

        stored_embedding = np.frombuffer(
            decrypted_data,
            dtype=np.float32
        )

        stored_embedding = torch.tensor(
            stored_embedding
        )

        # -----------------------------------
        # 6. Make embedding shapes identical
        # -----------------------------------

        login_embedding = login_embedding.reshape(
            1,
            -1
        )

        stored_embedding = stored_embedding.reshape(
            1,
            -1
        )

        # -----------------------------------
        # 7. Calculate similarity
        # -----------------------------------

        similarity = torch.nn.functional.cosine_similarity(
            login_embedding,
            stored_embedding,
            dim=1
        ).item()

        speaker_match = (
            similarity >=
            settings.VOICE_SIMILARITY_THRESHOLD
        )

        # -----------------------------------
        # 8. Final authentication decision
        # -----------------------------------

        if speaker_match:

            return {
                "authenticated": True,
                "liveness": True,
                "speaker_match": True,
                "message": "Voice authentication successful",
                "username": user.username,
                "spoken_text": spoken_text,
                "similarity": round(similarity, 4)
            }

        return {
            "authenticated": False,
            "liveness": True,
            "speaker_match": False,
            "message": "Speaker verification failed",
            "username": user.username,
            "spoken_text": spoken_text,
            "similarity": round(similarity, 4)
        }

    finally:

        # Delete temporary audio
        if os.path.exists(file_path):
            os.remove(file_path)

            if "processed_file_path" in locals():
                if os.path.exists(processed_file_path):
                    os.remove(processed_file_path)

@router.post("/")
async def transcribe(file: UploadFile = File(...)):

    allowed_types = [
        "audio/wav",
        "audio/mpeg",
        "audio/mp3",
        "audio/x-wav",
        "audio/mp4",
        "audio/webm"
    ]

    if file.content_type not in allowed_types:
        raise HTTPException(
            status_code=400,
            detail="Unsupported audio format"
        )

    os.makedirs("temp", exist_ok=True)

    file_path = os.path.join(
        "temp",
        file.filename
    )

    with open(file_path, "wb") as buffer:
        buffer.write(await file.read())

    try:

        result = transcribe_audio(file_path)

        return {
            "filename": file.filename,
            "language": result["language"],
            "text": result["text"]
        }

    finally:

        if os.path.exists(file_path):
            os.remove(file_path)

async def verify_liveness(
    challenge: str = Form(...),

):

    os.makedirs("temp", exist_ok=True)

    file_path = os.path.join(
        "temp",

                "liveness": True,
                "message": "Challenge passed",
                "spoken_text": spoken_text

        return {
            "liveness": False,
            "message": "Challenge failed",
            "spoken_text": spoken_text
        }

    finally:

        # Delete temporary audio
        if os.path.exists(file_path):
            os.remove(file_path)

def generate_challenge(language: str):

    if language not in CHALLENGES:
        language = "en"

    return random.choice(CHALLENGES[language])


def normalize_text(text: str):

    text = text.lower().strip()

    # Remove punctuation
    text = re.sub(r"[^\w\s]", "", text)

    # Remove extra spaces
    text = " ".join(text.split())

    return text


def verify_challenge(expected: str, actual: str):

    expected = normalize_text(expected)
    actual = normalize_text(actual)

    return expected == actual

import random
import hashlib

from datetime import datetime, timedelta

from app.config import settings


def generate_otp():

    otp = ""

    for _ in range(settings.OTP_LENGTH):
        otp += str(random.randint(0, 9))

    return otp


def hash_otp(otp: str):

    return hashlib.sha256(
        otp.encode()
    ).hexdigest()


def verify_otp(otp: str, otp_hash: str):

    return hash_otp(otp) == otp_hash


def get_otp_expiry():

    return datetime.utcnow() + timedelta(
        minutes=settings.OTP_EXPIRE_MINUTES
    )
